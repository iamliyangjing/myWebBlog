## FTP

```groovy
@Slf4j
class FtpClientStrategy implements FtpStrategy<FTPClient> {

    private IntegrationConf.FTPInfo ftpInfo

    private static final int BUFFER_SIZE = 1024 * 1024 * 10

    FtpClientStrategy(IntegrationConf.FTPInfo ftpInfo) {
        this.ftpInfo = ftpInfo
    }

    @Override
    FTPClient login() {
        FTPClient ftpClient = new FTPClient()
        IntegrationConf.FTPInfo ftpConf = ftpInfo
        try {
            ftpClient.connect(ftpConf.ip, ftpConf.port)
            ftpClient.login(ftpConf.userName, ftpConf.password)
            ftpClient.setFileType(FTPClient.BINARY_FILE_TYPE)
            int reply = ftpClient.getReplyCode()
            if (!FTPReply.isPositiveCompletion(reply)) {
                closeConnect(ftpClient)
                Panic.connectionError("FTP Connection Error: " + MyJsonUtils.toJson(ftpConf))
            }
        } catch (Exception e) {
            Panic.connectionError("FTP Connection Error: {}", e)
        }
        return ftpClient
    }

    @Override
    void closeConnect(FTPClient ftpClient) {
        if (ftpClient != null && ftpClient.isConnected()) {
            try {
                ftpClient.logout()
                ftpClient.disconnect()
            } catch (IOException e) {
                Panic.connectionError("Close FTP Error: ", e)
            }
        }
    }

    @Override
    void uploadFile(String ftpPath, String fileName, byte[] fileBytes) {
        FTPClient ftpClient = login()
        InputStream inputStream = new ByteArrayInputStream(fileBytes)
        String fullFtpPath = ftpPath + fileName
        try {
            ftpClient.enterLocalPassiveMode()
            ftpClient.storeFile(fullFtpPath, inputStream)
            log.info("upload file ${fullFtpPath} successfully")
        } catch (Exception e) {
            Panic.systemException("upload file ${fullFtpPath} failed ", e)
        } finally {
            inputStream.close()
            closeConnect(ftpClient)
        }
    }

    @Override
    void downloadFile(String ftpPath, String fileName, String savePath) {
        FTPClient ftpClient = login()
        try {
            String[] files = checkDirAndFile(ftpClient, ftpPath)
            String eqFile = files.find {
                it == fileName
            }
            File file = new File(savePath + '/' + eqFile)
            OutputStream os = new FileOutputStream(file)
            ftpClient.retrieveFile(fileName, os)
            log.info("download file ${fileName} successfully")
        } catch (Exception e) {
            log.info("download file ${fileName} failed")
            Panic.systemException("Download file error: ", e)
        } finally {
            closeConnect(ftpClient)
        }
    }

    private String[] checkDirAndFile(FTPClient ftpClient, String ftpPath) {
        String path = ftpPath
        if (!ftpClient.changeWorkingDirectory(path)) {
            Panic.defFileNotFound(ftpPath + "non directory")
        }
        ftpClient.enterLocalPassiveMode()
        String[] fs = ftpClient.listNames()
        if (fs == null || fs.length == 0) {
            Panic.defFileNotFound(ftpPath + " non file")
        }
        return fs
    }

    @Override
    Map<String, byte[]> getFileBytes(String ftpPath) {
        FTPClient ftpClient = login()
        Map<String, byte[]> map = new HashMap<String, byte[]>()
        if (ftpClient != null) {
            try {
                String[] files = checkDirAndFile(ftpClient, ftpPath)
                for (String ff : files) {
                    InputStream is = ftpClient.retrieveFileStream(ff)
                    ByteArrayOutputStream byteStream = new ByteArrayOutputStream()
                    byte[] buffer = new byte[BUFFER_SIZE]
                    int readLength
                    while ((readLength = is.read(buffer, 0, BUFFER_SIZE)) > 0) {
                        byteStream.write(buffer, 0, readLength)
                    }
                    map.put(ff, byteStream.toByteArray())
                    ftpClient.completePendingCommand()
                }
            } catch (Exception e) {
                Panic.systemException("Download file error: ", e)
            } finally {
                closeConnect(ftpClient)
            }
        }
        return map
    }

    @Override
    byte[] getFileBytesByName(String ftpPath, String fileName) {
        FTPClient ftpClient = login()
        ByteArrayOutputStream byteStream = new ByteArrayOutputStream()
        if (ftpClient != null) {
            try {
                String[] files = checkDirAndFile(ftpClient, ftpPath)
                for (String ff : files) {
                    int index = ff.indexOf(fileName)
                    if (index != -1) {
                        InputStream is = ftpClient.retrieveFileStream(ff)
                        byte[] buffer = new byte[BUFFER_SIZE]
                        int len
                        while ((len = is.read(buffer, 0, BUFFER_SIZE)) != -1) {
                            byteStream.write(buffer, 0, len)
                        }
                        break
                    }
                }
            } catch (Exception e) {
                Panic.systemException("Download file error: ${e.getMessage()} ", e)
            } finally {
                closeConnect(ftpClient)
            }
        }
        return byteStream.toByteArray()
    }

    @Override
    Map<String, InputStream> getFileInputStream(String ftpPath) {
        FTPClient ftpClient = login()
        Map<String, InputStream> map = new HashMap<String, InputStream>()
        if (ftpClient != null) {
            try {
                String[] files = checkDirAndFile(ftpClient, ftpPath)
                for (String ff : files) {
                    InputStream is = ftpClient.retrieveFileStream(ff)
                    map.put(ff, is)
                    ftpClient.completePendingCommand()
                }
            } catch (IOException e) {
                Panic.systemException("Download file error: ", e)
            } finally {
                closeConnect(ftpClient)
            }
        }
        return map
    }

    @Override
    InputStream getInputStreamByName(String ftpPath, String fileName) {
        FTPClient ftpClient = login()
        InputStream input = null
        if (ftpClient != null) {
            try {
                String[] files = checkDirAndFile(ftpClient, ftpPath)
                for (String ff : files) {
                    int index = ff.indexOf(fileName)
                    if (index != -1) {
                        input = ftpClient.retrieveFileStream(ff)
                        break
                    }
                }
            } catch (IOException e) {
                Panic.systemException("Download file error: ", e)
            } finally {
                closeConnect(ftpClient)
            }
        }
        return input
    }

    @Override
    boolean rename(String from, String to) {
        FTPClient ftpClient = login()
        boolean success = false
        try {
            success = ftpClient.rename(from, to)
        } catch (Exception e) {
            Panic.systemException("Move file error: ", e)
        } finally {
            closeConnect(ftpClient)
        }
        return success
    }

    @Override
    String[] listNames(String ftpPath, String fileName) {
        FTPClient ftpClient = login()
        String[] fileNames = null
        try {
            String fullFtpPath = MyStringUtils.isEmpty(fileName) ? ftpPath : ftpPath + fileName
            ftpClient.enterLocalPassiveMode()
            fileNames = ftpClient.listNames(fullFtpPath)
        } catch (Exception e) {
            Panic.systemException("list file names error: ", e)
        } finally {
            closeConnect(ftpClient)
        }
        return fileNames
    }

    @Override
    boolean deleteFile(String source) {
        FTPClient ftpClient = login()
        boolean success = false
        try {
            success = ftpClient.deleteFile(source)
        } catch (Exception e) {
            Panic.systemException("delete file error: ", e)
        } finally {
            closeConnect(ftpClient)
        }
        return success
    }
}
```

## IntegrationFtpUtils

IntegrationFtpUtils工具构建，使用哪一种协议工具类

```groovy
package com.bytesforce.insmate.pub.utils

import com.bytesforce.insmate.config.IntegrationConf
import com.bytesforce.insmate.pub.utils.ftp.FtpClientStrategy
import com.bytesforce.insmate.pub.utils.ftp.FtpStrategy
import com.bytesforce.insmate.pub.utils.ftp.SftpClientStrategy

//@Service
class IntegrationFtpUtils {

    private FtpStrategy ftpStrategy


    private IntegrationFtpUtils() {}

    static IntegrationFtpUtils build() {
        return new IntegrationFtpUtils()
    }

    IntegrationFtpUtils withFTP(IntegrationConf.FTPInfo ftpInfo) {
        if (ftpInfo.enableSSH) {
            this.ftpStrategy = new SftpClientStrategy(ftpInfo)
        } else {
            this.ftpStrategy = new FtpClientStrategy(ftpInfo)
        }
        return this
    }
    /**
     * download file to local
     * @param ftpPath ftp sever relative directory
     * @param fileName download's file name
     * @param savePath save to local directory
     */
    void downloadFile(String ftpPath, String fileName, String savePath) {
        ftpStrategy.downloadFile(ftpPath, fileName, savePath)
    }

    /**
     * Get all file's byte array in the ftp path
     * @param ftpPath
     * @return Map<String, byte[]> Key = fileName, Value = fileName's bytes
     */
    Map<String, byte[]> getFileBytes(String ftpPath) {
        return ftpStrategy.getFileBytes(ftpPath)
    }

    /**
     * Get fileName byte array in the ftp path
     * @param ftpPath
     * @param fileName
     * @return bytes
     */
    byte[] getFileBytesByName(String ftpPath, String fileName) {
        return ftpStrategy.getFileBytesByName(ftpPath, fileName)
    }

    /**
     * Get all InputStream in the ftp path
     * @param ftpPath
     * @return Map<String, byte[]> Key = fileName, Value = fileName's inputStream
     */
    Map<String, InputStream> getFileInputStream(String ftpPath) {
        return ftpStrategy.getFileInputStream(ftpPath)
    }

    /**
     * Get ftpPath InputStream in the ftp path
     * @param ftpPath
     * @return Map<String, byte[]> Key = fileName, Value = fileName's inputStream
     */
    InputStream getInputStreamByName(String ftpPath, String fileName) {
        return ftpStrategy.getInputStreamByName(ftpPath, fileName)
    }

    void uploadFile(String ftpPath, String fileName, byte[] fileBytes) {
        ftpStrategy.uploadFile(ftpPath, fileName, fileBytes)
    }

    boolean rename(String from, String to) {
        return ftpStrategy.rename(from, to)
    }

    boolean deleteFile(String source) {
        return ftpStrategy.deleteFile(source)
    }

    String[] listNames(String ftpPath, String fileName = null) {
        return ftpStrategy.listNames(ftpPath, fileName)
    }
}
```
